cmake_minimum_required(VERSION 3.13)
project(pico_wakecon_host_tests C)
set(CMAKE_C_STANDARD 11)
if(MSVC)
  add_compile_options(/utf-8 /W3)
endif()
include(CTest)
add_executable(test_util test_util.c ../../src/util.c ../../src/spi.c)
target_include_directories(test_util PRIVATE ../../src)
add_executable(test_cap test_cap.c ../../src/cap.c ../../src/util.c ../../src/spi.c)
target_include_directories(test_cap PRIVATE ../../src)
add_test(NAME util COMMAND test_util)
add_test(NAME cap COMMAND test_cap)
